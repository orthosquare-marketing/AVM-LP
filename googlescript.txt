/**
 * Lead capture endpoint for the AVM landing pages.
 *
 * Columns are matched by HEADER NAME, not by position, so manually added
 * columns (REMARK, LOCATION, ...) are never overwritten and any new field
 * is appended after whatever the sheet already has.
 */

var LEAD_FIELDS = [
  "timestamp",
  "date",
  "firstName",
  "lastName",
  "phone",
  "email",
  "pincode",
  "city",
  "treatment",
  "sheet_tab",
  "utm_source",
  "utm_medium",
  "utm_campaign",
  "utm_term",
  "utm_content",
  "gclid",
  "gbraid",
  "wbraid",
  "fbclid",
  "fbc",
  "fbp",
  "msclkid",
  "ttclid",
  "li_fat_id",
  "twclid",
  "referrer",
  "landing_page"
];

function doPost(e) {
  try {
    const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
    const params = normalizeParams_(e);

    const sheetName = sanitizeSheetName_(params.sheet_tab) || "Leads";
    const sheet = getOrCreateSheet_(spreadsheet, sheetName);

    const headers = ensureHeaders_(sheet, LEAD_FIELDS);

    const values = {
      timestamp: new Date(),
      date: valueOrBlank_(params.date),
      firstName: valueOrBlank_(params.firstName),
      lastName: valueOrBlank_(params.lastName),
      phone: valueOrBlank_(params.phone),
      email: valueOrBlank_(params.email),
      pincode: valueOrBlank_(params.pincode),
      city: valueOrBlank_(params.city),
      treatment: valueOrBlank_(params.treatment),
      sheet_tab: valueOrBlank_(params.sheet_tab),
      utm_source: valueOrBlank_(params.utm_source),
      utm_medium: valueOrBlank_(params.utm_medium),
      utm_campaign: valueOrBlank_(params.utm_campaign),
      utm_term: valueOrBlank_(params.utm_term),
      utm_content: valueOrBlank_(params.utm_content),
      gclid: valueOrBlank_(params.gclid),
      gbraid: valueOrBlank_(params.gbraid),
      wbraid: valueOrBlank_(params.wbraid),
      fbclid: valueOrBlank_(params.fbclid),
      fbc: valueOrBlank_(params.fbc),
      fbp: valueOrBlank_(params.fbp),
      msclkid: valueOrBlank_(params.msclkid),
      ttclid: valueOrBlank_(params.ttclid),
      li_fat_id: valueOrBlank_(params.li_fat_id),
      twclid: valueOrBlank_(params.twclid),
      referrer: valueOrBlank_(params.referrer),
      landing_page: valueOrBlank_(params.landing_page)
    };

    // One cell per existing column; anything the form doesn't own stays blank.
    const row = headers.map(function (header) {
      return Object.prototype.hasOwnProperty.call(values, header) ? values[header] : "";
    });

    sheet.appendRow(row);

    return ContentService
      .createTextOutput("ok")
      .setMimeType(ContentService.MimeType.TEXT);
  } catch (error) {
    return ContentService
      .createTextOutput("error: " + error.message)
      .setMimeType(ContentService.MimeType.TEXT);
  }
}

function doGet() {
  return ContentService
    .createTextOutput("Google Apps Script is live. v3-multi-platform-click-ids")
    .setMimeType(ContentService.MimeType.TEXT);
}

function normalizeParams_(e) {
  const params = (e && e.parameter) ? e.parameter : {};
  const normalized = {};

  Object.keys(params).forEach(function(key) {
    normalized[key] = typeof params[key] === "string" ? params[key].trim() : params[key];
  });

  return normalized;
}

function getOrCreateSheet_(spreadsheet, sheetName) {
  let sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) {
    sheet = spreadsheet.insertSheet(sheetName);
  }
  return sheet;
}

/**
 * Returns the sheet's header row, after restoring or appending any required
 * header that is missing. Existing headers keep their position and their name.
 *
 * A header cell that has been blanked out (a stray edit, a cleared cell) used to
 * make the column look missing, so the script appended a duplicate at the far
 * right and wrote the data there instead — leaving the original column empty.
 * A blank cell is refilled in place only when it sits at the column's own
 * position in LEAD_FIELDS *and* a neighbouring header agrees it belongs there.
 * Any other blank is left alone: it is somebody's spacer, not a lost column.
 */
function ensureHeaders_(sheet, requiredHeaders) {
  if (sheet.getLastRow() === 0 || sheet.getLastColumn() === 0) {
    sheet.getRange(1, 1, 1, requiredHeaders.length).setValues([requiredHeaders]);
    sheet.setFrozenRows(1);
    return requiredHeaders.slice();
  }

  let headers = sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getValues()[0]
    .map(function (header) {
      return String(header || "").trim();
    });

  const missing = requiredHeaders.filter(function (header) {
    return headers.indexOf(header) === -1;
  });

  if (!missing.length) {
    return headers;
  }

  const stillMissing = [];
  missing.forEach(function (header) {
    const canonical = requiredHeaders.indexOf(header);
    if (canonical < headers.length && headers[canonical] === "" &&
        neighbourAgrees_(headers, requiredHeaders, canonical)) {
      headers[canonical] = header;
      sheet.getRange(1, canonical + 1).setValue(header);
    } else {
      stillMissing.push(header);
    }
  });

  if (stillMissing.length) {
    sheet
      .getRange(1, headers.length + 1, 1, stillMissing.length)
      .setValues([stillMissing]);
    headers = headers.concat(stillMissing);
  }

  sheet.setFrozenRows(1);
  return headers;
}

/**
 * True when the column either side of `index` carries the header LEAD_FIELDS
 * expects there — evidence that the blank cell really is that column, and not
 * an unrelated gap that happens to land on the same index.
 */
function neighbourAgrees_(headers, requiredHeaders, index) {
  const before = index > 0 &&
    headers[index - 1] !== "" &&
    headers[index - 1] === requiredHeaders[index - 1];
  const after = index + 1 < headers.length &&
    headers[index + 1] !== "" &&
    headers[index + 1] === requiredHeaders[index + 1];
  return before || after;
}

function sanitizeSheetName_(name) {
  return String(name || "")
    .replace(/[\\/\?\*\[\]\:]/g, " ")
    .trim()
    .substring(0, 100);
}

function valueOrBlank_(value) {
  return value || "";
}
